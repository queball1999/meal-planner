{{/* Shared list of primary navigation links, rendered in two places with two class sets: the desktop header bar (Style "nav") and the mobile popover menu (Style "popover"). Call with: {{template "partials/nav_items.html" dict "Page" .Page "Style" "nav" "WeekParam" .WeekParam}} so the active-page highlight stays in one place instead of being kept in sync across two copies of six links. WeekParam is the week the *current* request explicitly asked to view (pageData.WeekParam - see requestWeekParam in render.go): only set when the URL that produced this page itself carried ?week= or ?calref=, never just because a plan happened to resolve to some other week. Carrying it into the Dashboard/Plan/Shopping List links keeps that same week in view across a tab switch instead of snapping back to today/latest; with no such param these stay the plain "/", "/plan", "/plan/list" links, which each page resolves to today/latest on its own. */}} {{define "partials/nav_items.html"}} {{- $page := .Page -}} {{- $link := "nav-link" -}} {{- $active := "nav-link--active" -}} {{- if eq .Style "popover" -}}{{- $link = "popover__item" -}}{{- $active = "popover__item--active" -}}{{- end -}} {{- $pantrySection := or (eq $page "pantry") (eq $page "items") (eq $page "item_detail") -}} {{- $financeSection := or (eq $page "finance_overview") (eq $page "finance_ai") (eq $page "finance_prices") -}} {{- $week := .WeekParam -}} {{- $dashHref := "/" -}} {{- $planHref := "/plan" -}} {{- $listHref := "/plan/list" -}} {{- if $week -}} {{- $dashHref = printf "/?calref=%s" $week -}} {{- $planHref = printf "/plan?week=%s" $week -}} {{- $listHref = printf "/plan/list?week=%s" $week -}} {{- end -}} {{icon "home"}} Dashboard {{icon "calendar"}} Plan {{icon "cart"}} Shopping List {{icon "archive"}} Pantry {{icon "fork"}} Recipes {{icon "history"}} Finance {{- if .IsAdmin}} {{icon "cog"}} Settings {{- end}} {{end}}
